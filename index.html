'block' : 'none'; document.getElementById('page-rebound').style.display = page === 'rebound' ? 'block' : 'none'; document.getElementById('page-archive').style.display = page === 'archive' ? 'block' : 'none'; if(page === 'archive') renderArchive(); } // SENSOR AKTIVIERUNG document.getElementById('reqBtn').onclick = () => { if (typeof DeviceMotionEvent.requestPermission === 'function') { DeviceMotionEvent.requestPermission().then(res => { if(res == 'granted') activate(); }); } else { activate(); } }; function activate() { document.getElementById('reqBtn').style.display = 'none'; document.getElementById('mainBtn').style.display = 'block'; window.addEventListener('devicemotion', (e) => { if (!isRecording) return; let acc = e.acceleration || e.accelerationIncludingGravity; let total = Math.sqrt(acc.x**2 + acc.y**2 + acc.z**2); // Wir ziehen die Erdanziehung ab, falls nötig let val = Math.abs(total - 9.81); if(val > 0.5) { // Filtert minimales Rauschen accelerationData.push(val); document.getElementById('mainBtn').innerText = `STOPP (${accelerationData.length})`; } }); } // AUFNAHME-LOGIK document.getElementById('mainBtn').onclick = function() { if (!isRecording) { accelerationData = []; isRecording = true; this.classList.add('record'); } else { isRecording = false; this.classList.remove('record'); this.innerText = "START AUFNAHME"; document.getElementById('setup-area').style.display = 'none'; document.getElementById('feedback-modal').style.display = 'block'; } }; function finalizeSave() { if(accelerationData.length < 10) return; // PEAK-ERKENNUNG (Wir suchen echte Schläge) let peaks = []; const threshold = 15; // Nur Werte über 15 m/s² sind echte Peaks for(let i = 1; i < accelerationData.length - 1; i++) { if(accelerationData[i] > threshold && accelerationData[i] > accelerationData[i-1] && accelerationData[i] > accelerationData[i+1]) { peaks.push({idx: i, val: accelerationData[i]}); } } const run = { id: Date.now(), date: new Date().toLocaleString('de-DE'), name: document.getElementById('runName').value || "Lauf", setup: { psi: document.getElementById('springVal').value, lsr: document.getElementById('lsrVal').value }, feedback: { grip: document.getElementById('rate-grip').value, arm: document.getElementById('rate-arm').value, msg: document.getElementById('run-comment').value }, data: accelerationData, peaks: peaks.sort((a,b) => b.val - a.val).slice(0, 10), avg: accelerationData.reduce((a,b) => a+b, 0) / accelerationData.length }; let runs = JSON.parse(localStorage.getItem('mtb_runs') || '[]'); runs.push(run); localStorage.setItem('mtb_runs', JSON.stringify(runs)); document.getElementById('feedback-modal').style.display = 'none'; document.getElementById('setup-area').style.display = 'block'; showPage('archive'); } // REBOUND-TEST (MIT FILTER) document.getElementById('reboundStartBtn').onclick = function() { this.innerText = "JETZT PLOPPEN LASSEN..."; this.classList.add('record'); accelerationData = []; isRecording = true; setTimeout(() => { isRecording = false; this.innerText = "Test Starten (3s)"; this.classList.remove('record'); processReboundTest(); }, 3000); }; function processReboundTest() { if(accelerationData.length < 5) return; // Filter: Wir zählen nur Hübe, die eine gewisse Energie haben let oscillations = 0; let lastPeakTime = 0; const minPower = Math.max(...accelerationData) * 0.3; // 30% des Max-Schlags for(let i = 1; i < accelerationData.length - 1; i++) { if(accelerationData[i] > minPower && accelerationData[i] > accelerationData[i-1] && accelerationData[i] > accelerationData[i+1]) { // Zeit-Abstand zwischen Schwingungen (Debounce) if(i - lastPeakTime > 5) { oscillations++; lastPeakTime = i; } } } // Der erste Peak ist der Bordstein selbst, den ziehen wir ab let realOsc = Math.max(0, oscillations - 1); let advice = ""; if(realOsc === 0) advice = "✅ Perfekt. Gabel steht sofort still."; else if(realOsc === 1) advice = "🟡 Gut. Ein kleiner Nachschwing-Hub."; else advice = `🐢 Zu schnell! (${realOsc} Nachschwingungen). Rebound langsamer stellen.`; document.getElementById('reboundResult').innerHTML = `Ergebnis: ${realOsc} Nachschwingungen.
${advice}`; } function renderArchive() { const list = document.getElementById('archive-list'); let runs = JSON.parse(localStorage.getItem('mtb_runs') || '[]'); list.innerHTML = ""; runs.forEach(run => { const item = document.createElement('div'); item.className = 'archive-item'; item.innerHTML = ` ${run.name}
${run.date} | PSI: ${run.setup.psi} | Grip: ${run.feedback.grip}/5 `; item.onclick = () => { renderSingleChart(run); }; list.appendChild(item); }); } function renderSingleChart(run) { const canvas = document.getElementById('myChart'); canvas.style.display = 'block'; if(chart) chart.destroy(); chart = new Chart(canvas.getContext('2d'), { type: 'line', data: { labels: run.data.map((_, i) => i), datasets: [{ label: run.name + ' (m/s²)', data: run.data, borderColor: '#007bff', borderWidth: 1.5, pointRadius: run.data.map((v, i) => { // Markiere Top-Peaks rot return run.peaks.some(p => p.idx === i) ? 5 : 0; }), pointBackgroundColor: 'red' }] }, options: { responsive: true, maintainAspectRatio: false, scales: { y: { beginAtZero: true } } } }); document.getElementById('compareDetails').innerHTML = `
Details: Ø ${run.avg.toFixed(1)} m/s² | Max Peak: ${run.peaks[0]?.val.toFixed(1) || 0}
"${run.feedback.msg || 'Kein Kommentar'}"
`; } function deleteRun(id) { let runs = JSON.parse(localStorage.getItem('mtb_runs') || '[]'); localStorage.setItem('mtb_runs', JSON.stringify(runs.filter(r => r.id !== id))); renderArchive(); }
